#include <iostream>
#include <stdexcept>
using namespace std;

int dividir(int a, int b) {
    if (b == 0) {
        throw runtime_error("Error de ejecucion entre cero");
    }
    return a / b;
}

int main() {
    try {
        cout << dividir(10, 0) << endl;
    }
    catch (const exception& e)
    {
        cout << "Error: " << e.what();
    }
    
}
